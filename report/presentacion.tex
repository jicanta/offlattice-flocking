\section{Modelo}

\begin{frame}{Modelo de Vicsek}
    % Contenido.
    Actualización sincrónica: la posición \textbf{x}$_i$ avanza con la
    velocidad \textbf{v}$_i$ del paso anterior, y el ángulo $\theta_i$
    incorpora ruido uniforme sobre el promedio de orientaciones vecinas,
    Ec.~(\ref{eq:posicion}) y Ec.~(\ref{eq:angulo}).
    \begin{equation}
        \mathbf{x}_i(t+1) = \mathbf{x}_i(t) + \mathbf{v}_i(t)\, dt
        \label{eq:posicion}
    \end{equation}
    \begin{equation}
        \theta_i(t+1) = \operatorname{base}_i(t) + \mathcal{U}\!\left[-\frac{\eta}{2},\ \frac{\eta}{2}\right]
        \label{eq:angulo}
    \end{equation}
    donde $\eta$ es la amplitud de ruido y $\operatorname{base}_i(t)$ el
    promedio vectorial de las orientaciones dentro del radio $r_c$.
\end{frame}
