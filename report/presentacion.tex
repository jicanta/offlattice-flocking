\section{Introducción}
% =========================================================

\begin{frame}{Sistema real}
    \vspace{1.4em}
    \begin{itemize}
        \setlength{\itemsep}{1.5em}
        \item Bandadas, cardúmenes, colonias bacterianas
        \item Individuos que se \alert{autopropulsan} y se \alert{alinean}
              con sus vecinos
        \item Interacción \alert{local}, orden colectivo \alert{espontáneo}
    \end{itemize}
    \vspace{2.0em}
    \cita{Vicsek et al., Phys. Rev. Lett. 75, 1226, 1995}
\end{frame}

\begin{frame}{Modelo matemático}
    \vspace{0.2em}
    {\small $N$ partículas puntuales, rapidez común $v$; actualización
    \alert{sincrónica}}
    \vspace{-0.2em}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.45\textwidth}
            \begin{equation*}
                \mathbf{r}_i(t+1) = \mathbf{r}_i(t) + \mathbf{v}_i(t)\,\Delta t
            \end{equation*}
        \end{column}
        \begin{column}{0.53\textwidth}
            \begin{equation*}
                \theta_i(t+1) = \Theta_i(t) + \Delta\theta_i,
                \quad
                \Delta\theta_i \sim \mathcal{U}\!\left[-\tfrac{\eta}{2},\tfrac{\eta}{2}\right]
            \end{equation*}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    {\small $\mathcal{N}_i=\{\,j\neq i:\ d(\mathbf{r}_i,\mathbf{r}_j)<r_c\,\}$;
    \ $\eta$ en radianes}
    \vspace{1.2em}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            {\color{azul}\bfseries Estándar}
            \begin{equation*}
                \Theta_i = \arctan
                \frac{\sin\theta_i + \sum_{j\in\mathcal{N}_i}\sin\theta_j}
                     {\cos\theta_i + \sum_{j\in\mathcal{N}_i}\cos\theta_j}
            \end{equation*}
            \vspace{-0.7em}
            \begin{center}\cita{Vicsek et al., PRL, 1995}\end{center}
        \end{column}\hfill
        \begin{column}{0.48\textwidth}
            {\color{azul}\bfseries Votante}
            \begin{equation*}
                \Theta_i = \theta_k,
                \qquad
                k \sim \mathcal{U}\!\left(\mathcal{N}_i\cup\{i\}\right)
            \end{equation*}
            \vspace{-0.7em}
            \begin{center}\cita{Loscar et al., Phys. Rev. E 104, 034111, 2021}\end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Del observable temporal al escalar}
    \figpar[0.64]{b_evolucion_temporal/va_S_vs_t_vicsek_rho4_eta2_M20.png}{%
        \setup{Condiciones}
        Modelo estándar\\
        $\rho = \SI{4}{m^{-2}}$ ($N = 400$), $\eta = \SI{2}{rad}$\\
        \SI{5e3}{s} por corrida
        \vspace{0.9em}

        \setup{\teq}
        Primer bloque de \SI{50}{s} de la curva \alert{promedio} que cruza la
        media de la segunda mitad
        \vspace{0.9em}

        \setup{Escalar y barra}
        \alert{Todas} las muestras con $t \ge \teq$, de las $M$ realizaciones}
    \vspace{0.3em}
    {\scriptsize
    $\displaystyle
    \langle X\rangle=\frac{1}{MT}\sum_{m=1}^{M}\sum_{t\ge\teq}X_m(t),
    \qquad
    \sigma_X=\sqrt{\frac{1}{MT-1}\sum_{m=1}^{M}\sum_{t\ge\teq}\big(X_m(t)-\langle X\rangle\big)^2},
    \qquad X\in\{\va,\,S\}$}
\end{frame}

\section{Conclusiones}

\begin{frame}{Conclusiones}
    \vspace{-0.2em}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}\small
            \begin{itemize}\setlength{\itemsep}{0.9em}
                \item Estándar: \alert{transición continua} controlada por
                      $\eta$
                \item \alert{La densidad favorece el orden}:
                      $\eta(\va = 0{,}5) = \SI{2.9}{rad} \to \SI{3.5}{rad}$
                \item Votante: \alert{un orden más sensible}
                      ($\eta(\va = 0{,}5) \simeq \SI{0.3}{rad}$), densidad con
                      \alert{signo opuesto}
                \item En $\eta = \SI{5}{rad}$ el votante queda desordenado
                      por completo; el estándar todavía no
            \end{itemize}
        \end{column}\hfill
        \begin{column}{0.48\textwidth}\small
            \begin{itemize}\setlength{\itemsep}{0.9em}
                \item Con $\rho = 2$, $4$, $\SI{8}{m^{-2}}$: $S \ge 0{,}90$
                      siempre, \alert{no es un grado de libertad}
                \item A $1/\pi$ y $1/3\pi$: $S$ \alert{acoplado} a $\va$;
                      efecto de los parámetros, no del modelo
                \item Sin ruido percola con un vecino por partícula:
                      \alert{fragmenta el ruido}
                \item CIM $\times\num{10}$ sobre la fuerza bruta en $N = 1150$,
                      con el escalamiento del TP1
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
