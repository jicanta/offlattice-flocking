\section{Introducción}
% =========================================================

\begin{frame}{Sistema real}
    \vspace{1.4em}
    \begin{itemize}
        \setlength{\itemsep}{1.5em}
        \item Bandadas, cardúmenes, colonias bacterianas
        \item Individuos que se \alert{autopropulsan} y se \alert{alinean}
              con sus vecinos
        \item Interacción \alert{local}, orden colectivo \alert{espontáneo}
    \end{itemize}
    \vspace{2.0em}
    \cita{Vicsek et al., Phys. Rev. Lett. 75, 1226, 1995}
\end{frame}

\begin{frame}{Modelo matemático}
    \vspace{0.2em}
    {\small $N$ partículas puntuales, rapidez común $v$; actualización
    \alert{sincrónica}}
    \vspace{-0.2em}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.45\textwidth}
            \begin{equation*}
                \mathbf{r}_i(t+1) = \mathbf{r}_i(t) + \mathbf{v}_i(t)\,\Delta t
            \end{equation*}
        \end{column}
        \begin{column}{0.53\textwidth}
            \begin{equation*}
                \theta_i(t+1) = \Theta_i(t) + \Delta\theta_i,
                \quad
                \Delta\theta_i \sim \mathcal{U}\!\left[-\tfrac{\eta}{2},\tfrac{\eta}{2}\right]
            \end{equation*}
        \end{column}
    \end{columns}
    \vspace{0.3em}
    {\small $\mathcal{N}_i=\{\,j\neq i:\ d(\mathbf{r}_i,\mathbf{r}_j)<r_c\,\}$}
    \vspace{1.2em}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            {\color{azul}\bfseries Estándar}
            \begin{equation*}
                \Theta_i = \arctan
                \frac{\sin\theta_i + \sum_{j\in\mathcal{N}_i}\sin\theta_j}
                     {\cos\theta_i + \sum_{j\in\mathcal{N}_i}\cos\theta_j}
            \end{equation*}
            \vspace{-0.7em}
            \begin{center}\cita{Vicsek et al., PRL, 1995}\end{center}
        \end{column}\hfill
        \begin{column}{0.48\textwidth}
            {\color{azul}\bfseries Votante}
            \begin{equation*}
                \Theta_i = \theta_k,
                \qquad
                k \sim \mathcal{U}\!\left(\mathcal{N}_i\cup\{i\}\right)
            \end{equation*}
            \vspace{-0.7em}
            \begin{center}\cita{Loscar et al., Phys. Rev. E 104, 034111, 2021}\end{center}
        \end{column}
    \end{columns}
\end{frame}

\section{Conclusiones}
% =========================================================

\begin{frame}{Conclusiones}
    \begin{itemize}\setlength{\itemsep}{1.4em}
        \item El ruido destruye el orden en las dos reglas; el votante lo
              pierde con un ruido diez veces menor.
        \item La densidad ordena al estándar y desordena al votante:
              promediar atenúa el ruido, copiar a uno solo no.
        \item Con $\rho = 2$, $4$ y $\SI{8}{m^{-2}}$ el sistema está siempre
              percolado ($S \simeq 1$); a baja densidad $S$ cae con el ruido y
              $\va$ sigue a $S$.
        \item Sin ruido los grupos no se deshacen: fragmenta el ruido, no la
              densidad.
        \item CIM: diez veces más rápido que la fuerza bruta en $N = 1150$,
              con la misma pendiente que en el TP1.
    \end{itemize}
\end{frame}
